function textLabel(name,text,x,y,z,size,mat='DinoWeb Ink'){
 mesh('Text',{text,font:'Space Grotesk_regular',fontSize:size,depth:0.3});
 geometry({width:Math.max(160,text.length*size*.7),height:size*2.5});
 position({x,y,z});
 applyMaterial(mat);
 rename(name);
}
function box(name,w,h,d,x,y,z,mat='DinoWeb Limestone'){
 mesh('Cube',{width:w,height:h,depth:d,cornerRadius:0.5});
 position({x,y,z});
 applyMaterial(mat);
 rename(name);
}
